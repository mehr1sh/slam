\begin{frame}{Camera Model}
  Pinhole camera, intrinsic matrix $K$ (TUM Freiburg2 calibration used throughout):
  \[
    K =
    \begin{pmatrix}
      f_x & 0 & c_x \\
      0 & f_y & c_y \\
      0 & 0 & 1
    \end{pmatrix}
    =
    \begin{pmatrix}
      520.9 & 0 & 325.1 \\
      0 & 521.0 & 249.7 \\
      0 & 0 & 1
    \end{pmatrix}
  \]

  Pixel $\to$ normalized camera coordinates (\texttt{pixel2cam()}):
  \[
    x = K^{-1} p
  \]
\end{frame}

\begin{frame}{Two-View Geometry}
  \begin{center}
    \texttt{Matched points} $\Downarrow$\\
    \texttt{Fundamental Matrix} $\Downarrow$\\
    \texttt{Essential Matrix} $\Downarrow$\\
    \texttt{Recover Pose (R, t)}
  \end{center}

  \vspace{1em}
  \begin{itemize}
    \item 79 matches used
    \item Epipolar constraint $y_2^\top [t]_\times R\, y_1 \approx 0$ for every match,
          residuals $\sim 10^{-3}$ --- confirms $R, t$ are consistent
    \item $t$ is a \textbf{unit vector}: monocular two-view geometry has
          scale ambiguity, direction only
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Triangulation}
  \begin{center}
    \texttt{Two-view correspondences} $\Downarrow$\\
    \texttt{Projection matrices } $P_1 = K[I|0]$\texttt{, } $P_2 = K[R|t]$ $\Downarrow$\\
    \texttt{Triangulated 3D points}
  \end{center}

  Points kept only if positive depth in \emph{both} cameras (cheirality check).
  Depth is in the same unit-norm-$t$ scale as the two-view pose, not metres.

\begin{frame}{PnP}
  \begin{center}
    \texttt{3D points + 2D observations} $\Downarrow$\\
    \texttt{PnP (OpenCV solvePnP)} $\Downarrow$\\
    \texttt{Camera pose} $\Downarrow$\\
    \texttt{Gauss-Newton refinement} $\Downarrow$\\
    \texttt{g2o refinement}
  \end{center}

  \begin{itemize}
    \item 75 of 79 matches have valid depth in frame 1 $\to$ 3D-2D pairs
    \item Translation here is \textbf{metric} (from real depth, not unit-norm)
    \item Cost/chi\textsuperscript{2} convergence (both solvers, same optimum):
  \end{itemize}
  \[
    40518 \;\to\; 411 \;\to\; 299.76 \;\to\; 299.76
  \]
  All three methods (solvePnP, Gauss-Newton, g2o) agree to 6 significant figures.
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{ICP}
  \begin{center}
    \texttt{3D-3D correspondences} $\Downarrow$\\
    \texttt{Closed-form SVD (R, t)} $\Downarrow$\\
    \texttt{g2o bundle-adjustment refinement}
  \end{center}

  \begin{itemize}
    \item 72 of 79 matches have valid depth in \emph{both} frames
    \item Translation is metric (real depth in both point clouds)
    \item chi\textsuperscript{2}: $1.8161 \to 1.8155$ --- the closed-form SVD
          solution is already at (or very near) the optimum, so BA has
          little left to improve
    \item Check: $R\,p_2 + t \approx p_1$ verified on sample points
  \end{itemize}
\end{frame}
