cmake_minimum_required(VERSION 3.16)
project(corrected_rgbd_icp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV QUIET COMPONENTS core imgcodecs)
if(NOT OpenCV_FOUND)
  find_path(OpenCV_INCLUDE_DIRS opencv2/core.hpp PATH_SUFFIXES opencv4 REQUIRED)
  foreach(component core imgcodecs)
    find_library(OPENCV_${component} opencv_${component} REQUIRED)
    list(APPEND OpenCV_LIBS ${OPENCV_${component}})
  endforeach()
endif()
find_package(Eigen3 REQUIRED)
include(../cmake/feature_core.cmake)
add_library(rgbd_icp ../src/tracking/rgbd_icp.cpp)
target_include_directories(rgbd_icp PUBLIC ../include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(rgbd_icp PUBLIC Eigen3::Eigen ${OpenCV_LIBS} feature_core)
target_compile_options(rgbd_icp PRIVATE -Wall -Wextra -Wpedantic)
add_executable(icp_sequence ../apps/icp_sequence.cpp)
target_link_libraries(icp_sequence PRIVATE rgbd_icp)
add_executable(rgbd_icp_test ../tests/unit/rgbd_icp_test.cpp ../src/tracking/icp.cpp)
target_link_libraries(rgbd_icp_test PRIVATE rgbd_icp)
enable_testing()
add_test(NAME rgbd_icp_geometry COMMAND rgbd_icp_test)

find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME blender_data_bridge
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../tests/unit/test_blender_data.py)
endif()

include(../cmake/orb_benchmark.cmake)
